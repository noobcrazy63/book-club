function initLiveSync() {
database.ref('bookCatalog').on('value', snapshot => {
const data = snapshot.val();
if (data) {
bookDatabase = data;
} else {
bookDatabase = [...defaultBooks];
database.ref('bookCatalog').set(defaultBooks);
}
bookDatabase.sort((a, b) => a.title.localeCompare(b.title));
applyFilterAndRender();
if (currentUserName) updateDashboardProgressUI();


});
database.ref('users').on('value', snapshot => {
const usersData = snapshot.val();
renderLeaderboard(usersData);
if (currentUserName && snapshot.hasChild(currentUserName)) {
const userSnapshot = snapshot.child(currentUserName).val();
totalPagesRead = userSnapshot.totalPages || 0;
totalMinutesRead = userSnapshot.totalMinutes || 0;
selectedBooks = userSnapshot.selectedBooks || {};
totalPagesDisplay.innerText = totalPagesRead;
totalMinutesDisplay.innerText = totalMinutesRead;
updateDashboardProgressUI();
updatePersonalGoalUI();
renderAchievements();
}
});
if (currentUserName) {
loadDashboard();
}
}
function handleSearchSearch() {
currentPage = 1;
applyFilterAndRender();
}
function applyFilterAndRender() {
const query = catalogSearch.value.toLowerCase().trim();
if (query === "") {
filteredBooks = [...bookDatabase];
} else {
filteredBooks = bookDatabase.filter(book => book.title.toLowerCase().includes(query));
}
renderCatalogPage();
}
function renderCatalogPage() {
catalogGrid.innerHTML = "";


const totalPagesCount = Math.ceil(filteredBooks.length / booksPerPage) || 1;
if (currentPage > totalPagesCount) currentPage = totalPagesCount;
const startIndex = (currentPage - 1) * booksPerPage;
const endIndex = startIndex + booksPerPage;
const pageItems = filteredBooks.slice(startIndex, endIndex);
if (pageItems.length === 0) {
catalogGrid.innerHTML = `<div style="grid-column: 1/-1; text-align: center; color:
var(--muted-text); padding: 30px;">No books match your search criteria.</div>`;
} else {
pageItems.forEach((book) => {
const originalIndex = bookDatabase.findIndex(b => b.title === book.title);
const card = document.createElement('div');
card.classList.add('book-card');
card.innerHTML = `
<div class="book-info-container">
<div>
<div class="book-title">${book.title}</div>
<div class="book-author">by ${book.author}</div>
</div>
<div class="status-container">
<span class="stock-status ${book.inStock ? 'in' : 'out'}">${book.inStock ? 'In
Stock' : 'Out of Stock'}</span>
${book.inStock ? `<button class="choose-btn"
onclick="chooseBook(${originalIndex})">Choose</button>` : ''}
</div>
</div>
`;
catalogGrid.appendChild(card);
});
}
pageIndicator.innerText = `${currentPage} out of ${totalPagesCount} pages`;
prevPageBtn.disabled = currentPage === 1;
nextPageBtn.disabled = currentPage === totalPagesCount;
}
function goToPrevPage() {
if (currentPage > 1) {
currentPage--;
renderCatalogPage();


}
}
function goToNextPage() {
const totalPagesCount = Math.ceil(filteredBooks.length / booksPerPage) || 1;
if (currentPage < totalPagesCount) {
currentPage++;
renderCatalogPage();
}
}
function renderLeaderboard(usersData) {
leaderboardList.innerHTML = "";
let profiles = [];
if (usersData) {
Object.keys(usersData).forEach(name => {
profiles.push({
name: name,
pages: usersData[name].totalPages || 0
});
});
}
if (currentUserName && !profiles.some(p => p.name === currentUserName)) {
profiles.push({ name: currentUserName, pages: totalPagesRead });
}
profiles.sort((a, b) => b.pages - a.pages);
profiles.forEach((profile, index) => {
const li = document.createElement('li');
li.classList.add('leaderboard-item');
if (index === 0 && profile.pages > 0) {
li.classList.add('top-reader');
}
li.innerHTML = `
<span>${index + 1}. <strong>${profile.name}</strong> ${index === 0 &&
profile.pages > 0 ? ' ' : ''}</span>
<span>${profile.pages} pages</span>
`;
leaderboardList.appendChild(li);
});
}

modalForm.addEventListener('submit', function(e) {
e.preventDefault();
const title = document.getElementById('modal-title').value.trim();
const author = document.getElementById('modal-author').value.trim();
const pages = parseInt(document.getElementById('modal-pages').value);
if (title && author && !isNaN(pages)) {
bookDatabase.push({
title: title,

pages: pages,
inStock: true
});
bookDatabase.sort((a, b) => a.title.localeCompare(b.title));

function chooseBook(index) {
const book = bookDatabase[index];
if (!selectedBooks[book.title]) {
selectedBooks[book.title] = { index: index, read: 0, total: book.pages };
book.inStock = false;
saveProfileToCloud();
saveCatalogToCloud();
}
}
/* FIX: Looks up the book target directly by its string Title key inside selectedBooks */


function turnInBook(bookTitle) {
if (selectedBooks[bookTitle]) {
// Find the original object location in the master catalog array to restock it
const bookInCatalog = bookDatabase.find(b => b.title === bookTitle);
if (bookInCatalog) {
bookInCatalog.inStock = true;
}
delete selectedBooks[bookTitle];
saveProfileToCloud();
saveCatalogToCloud();
const entry = document.createElement('div');
entry.classList.add('log-entry');
entry.style.borderLeftColor = "#e63946";
entry.innerHTML = `<span>
Returned: <strong>"${bookTitle}"</strong> has been
turned back in to the library!</span>`;
logHistory.prepend(entry);
}
}

function updateDashboardProgressUI() {
progressContainer.innerHTML = "";
logBookSelect.innerHTML = '<option value="" disabled selected>Choose book to
update...</option>';
const keys = Object.keys(selectedBooks);
if (keys.length === 0) {
trackingBox.classList.add('hidden');
return;
}
trackingBox.classList.remove('hidden');
keys.forEach(title => {
const data = selectedBooks[title];
const percentage = Math.min(Math.round((data.read / data.total) * 100), 100);
// Escapes single quotes in book titles safely for the inline JavaScript call
const safeTitleString = title.replace(/'/g, "\\'");
const progItem = document.createElement('div');
progItem.classList.add('progress-item');
progItem.innerHTML = `


<div class="progress-header-row">
<div class="progress-labels">
<span class="progress-book-name">${title}</span>
<span class="progress-book-stats">${data.read} / ${data.total} pages completed
(${percentage}%)</span>
</div>
<button class="turn-in-btn" onclick="turnInBook('${safeTitleString}')">Turn In
Book</button>
</div>
<div class="progress-bar-bg">
<div class="progress-bar-fill" style="width: ${percentage}%"></div>
</div>
`;
progressContainer.appendChild(progItem);
const opt = document.createElement('option');
opt.value = title;
opt.innerText = title;
logBookSelect.appendChild(opt);
});
}
