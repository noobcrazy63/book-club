async function generateAvatar(prompt) { const loading = document.getElementById('avatar-loading'); const errorDiv = document.getElementById('avatar-error'); const submitBtn = document.getElementById('avatar-submit-btn'); const preview = document.getElementById('avatar-preview'); try { loading.classList.remove('hidden'); submitBtn.disabled = true; errorDiv.style.display = 'none'; // Use DiceBear API - free avatar generator, no API key needed // Generate a unique seed from the prompt const seed = prompt.replace(/\s+/g, '-').toLowerCase().substring(0, 20); const avatarUrl = `https://api.dicebear.com/7.x/avataaars/svg?seed=${seed}`; // Verify the image loads const img = new Image(); img.onload = () => { preview.innerHTML = ``; }; img.onerror = () => { throw new Error('Failed to generate avatar image'); }; img.src = avatarUrl; } catch (error) { errorDiv.textContent = 'Error: ' + error.message; errorDiv.style.display = 'block'; console.error('Avatar generation error:', error); } finally { loading.classList.add('hidden'); submitBtn.disabled = false; } }
